### 윈도우 함수 실습

- sum()

```sql
--이전 2개의 행부터 현재까지의 합계
select employee_id, salary, sum(salary) over (order by salary rows 2 preceding) as salary_sum
from employees;

--현재 행부터 이후의 값 합계
select employee_id, salary, sum(salary) over (order by salary rows between current row and unbounded following) as future_sum
from employees;

--현재 행 이후 두 개의 행까지 합계
select employee_id, salary, sum(salary) over (order by salary rows between current row and 2 following) as sum_next_2
from employees;
```

- 복합 옵션

```sql
--복합 옵션으로 윈도우 설정
select employee_id, department_id, salary, sum(salary) over (partition by department_id order by salary rows unbounded preceding) as cum_salary
from employees;

--고객별 누적잔액을 계산하여 출력
select customer_id, account_id, balance, sum(balance) over (partition by customer_id order by account_id) as cum_balance
from accounts;

--지점 별 최고 대출 금액 계산
select branch_id, loan_id, amount, max(amount) over (partition by branch_id) as max_loan_amount
from loans;

select status, loan_id, count(loan_id) over (partition by status) as status_count
from loans;
```

- ROWS와 RANGE는 용도에 따라 신중히 선택
- 데이터 셋이 클 경우 성능에 유의
- UNBOUNDED PRECEDING, FOLLOWING 옵션의 범위를 명확히 이해

- 실습과제

```sql
--거래 테이블에서 계좌별 누적 거래 금액을 계산하시오.
select account_id, amount, sum(amount) over (partition by account_id order by transaction_date rows unbounded preceding) cum_sum
from transactions;

--거래 테이블과 계좌별 최고 거래 금액을 출력하시오.
select account_id, amount, max(amount) over(partition by account_id) max_amount, round(amount/(max(amount) over(partition by account_id)),2) rate
from transactions;

--대출 테이블에서 지점별 누적 대출 금액을 계산하시오.
select branch_id, amount, sum(amount) over(partition by branch_id order by loan_date rows unbounded preceding) cum_amount
from loans; 

--계좌 테이블에서 고객별 평균 잔액을 계산하시오.(반올림하여 소수점 둘째 자리까지 출력)
select customer_id,account_id, balance, round(avg(balance) over (partition by customer_id),2) avg_balance
from accounts;
```

### 계층 쿼리

- 데이터가 부모-자식 관계로 구성된 계층적 구조를 조회하기 위한 쿼리
- 동작 원리
    - START WITH: 계층 구조의 시작점 지정
    - CONNECT BY: 부모-자식 관계 정의
    - PRIOR: 방향 설정
    - ORDER SIBLINGS BY: 동일 레벨 정렬
    - LEVEL: 깊이를 나타내는 가상 컬럼

```sql
SELECT EMPLOYEE_ID, NAME, LEVEL
FROM EMPLOYEES
START WITH MANAGER_ID IS NULL
CONNECT BY PRIOR EMPLOYEE_ID = MANAGER_ID;

--is null인 이유는 최상위(회사)는 manager_id가 없기 때문
```

### 전체 복습 문제

```sql
--1. 직무 별 상위 3명의 직원
select * from
(select job_id 직무, employee_id 사번, name 이름, salary 급여, rank() over(partition by job_id order by job_id, salary nulls last) 순위
from employees)t
where t.순위 <= 3;

--2. 직원 입사 연도 별 총 급여 및 누적 급여
select to_char(hire_date,'YYYY') from employees;

select hire_year, max(yearly_sal) yearly_total_salary, sum(cum_sal) cum_sal 
from
    (select to_char(hire_date,'YYYY') hire_year, 
    sum(salary) over(partition by to_char(hire_date,'YYYY')) yearly_sal, 
    sum(salary) over(partition by to_char(hire_date,'YYYY') order by hire_date rows unbounded preceding) cum_sal
    from employees)t
group by t.hire_year;

-- 3 직원 별 급여 순위와 부서 평균 급여 비교
SELECT EMPLOYEE_ID 사번, NAME 이름, SALARY 급여,
    RANK() OVER (ORDER BY SALARY NULLS LAST) "급여 순위",
    AVG(SALARY) OVER (PARTITION BY DEPARTMENT_ID) "부서 평균 급여",
    CASE
        WHEN SALARY > AVG(SALARY) OVER (PARTITION BY DEPARTMENT_ID) THEN '높다'
        ELSE '낮다'
    END "부서 내 급여"
FROM EMPLOYEES;

-- 4. 최근 1년간 대출을 받은 고객 중 대출 금액 상위 5명=
SELECT *
FROM
    (SELECT
        (SELECT NAME FROM CUSTOMERS C WHERE C.CUSTOMER_ID = L.CUSTOMER_ID) AS NAME,
        SUM(AMOUNT) OVER(PARTITION BY CUSTOMER_ID
        ORDER BY LOAN_DATE) AS TOTAL_AMOUNT
    FROM LOANS L
    WHERE MONTHS_BETWEEN(SYSDATE, LOAN_DATE) <= 12
    ORDER BY TOTAL_AMOUNT DESC)
WHERE ROWNUM <= 5;

-- 5. 이동 평균 (Moving Average)
SELECT *
FROM
    (SELECT
        EMPLOYEE_ID, NAME, SALARY,
        ROUND(AVG(SALARY) OVER(ORDER BY EMPLOYEE_ID
        ROWS 2 PRECEDING),2) AS AVG_SALARY
    FROM EMPLOYEES)
WHERE ROWNUM <=3
;

-- 6번
SELECT
    (SELECT NAME FROM CUSTOMERS C WHERE C.CUSTOMER_ID=L.CUSTOMER_ID) NAME,
    AMOUNT AS AMT, '대출' AS TYPE_
FROM
    (SELECT CUSTOMER_ID, AMOUNT, AVG(AMOUNT) OVER() AVG_AMT
    FROM LOANS
    WHERE STATUS = 'APPROVED')L
WHERE AMOUNT > AVG_AMT
UNION ALL
SELECT
    (SELECT NAME FROM CUSTOMERS C WHERE C.CUSTOMER_ID=A.CUSTOMER_ID),
    BALANCE, '계좌' AS TYPE_
FROM
    (SELECT CUSTOMER_ID, BALANCE, AVG(BALANCE) OVER() AVG_AMT
    FROM ACCOUNTS)A
WHERE BALANCE > AVG_AMT
;

-- 7번
SELECT * FROM (
SELECT DISTINCT JOB_ID "직무 ID",
    (SELECT DEPARTMENT_NAME FROM DEPARTMENTS WHERE E.DEPARTMENT_ID = DEPARTMENT_ID) "부서 이름",
    ROUND(AVG(SALARY) OVER (PARTITION BY DEPARTMENT_ID)) "직무별 평균 급여",
    ROUND(AVG(SALARY) OVER (PARTITION BY JOB_ID)) "부서별 평균 급여"
FROM EMPLOYEES E)
WHERE "직무별 평균 급여" >= "부서별 평균 급여"
    AND "부서 이름" IS NOT NULL;
    
    
--8번 (1)
SELECT DISTINCT BRANCH_ID 지점, ROUND(APPROVED_CNT/TOTAL_CNT,2) 대출승인율
FROM 
(
        SELECT 
            BRANCH_ID,
            (SELECT COUNT(*) FROM LOANS T2 WHERE STATUS='APPROVED' AND T1.BRANCH_ID = T2.BRANCH_ID GROUP BY BRANCH_ID ) AS APPROVED_CNT,
            COUNT(LOAN_ID) OVER(PARTITION BY BRANCH_ID) AS TOTAL_CNT
        FROM LOANS T1
);

--8번(2)
SELECT BRANCH_ID, 
    ROUND(
        COUNT(
            CASE STATUS
                WHEN 'APPROVED' THEN 1
                ELSE NULL
            END
        )/COUNT(*)
    ,2)
FROM LOANS
GROUP BY BRANCH_ID
ORDER BY BRANCH_ID;

-- 9번
SELECT *
FROM
    (SELECT
        NAME, JOB_ID, SALARY,
        RANK() OVER(PARTITION BY JOB_ID ORDER BY SALARY) AS RANK_
    FROM EMPLOYEES
    ORDER BY JOB_ID, SALARY DESC NULLS LAST)
ORDER BY JOB_ID;

-- 10번
SELECT (SELECT DEPARTMENT_NAME FROM DEPARTMENTS WHERE DEPARTMENT_ID = E.DEPARTMENT_ID) "부서 이름",
    COUNT(MANAGER_ID) "상사 있음", COUNT(*) - COUNT(MANAGER_ID) "상사 없음"
FROM EMPLOYEES E
WHERE DEPARTMENT_ID IS NOT NULL
GROUP BY DEPARTMENT_ID;
```

### pure jdbc

- SQL과 자바가 하나의 파일에 공존(DAO)
- 별도 라이브러리, 프레임워크를 사용(MyBatis)
    - ORM(Obejct Related Mapping)으로 클래스 객체를 통해 DB 테이블을 쉽게 핸들링할 수 있음

### MyBatis

- SQL 구문과 자바 코드를 분리
- SQL의 실행결과를 Map에 매핑시킴
- DataSource, Transaction 등의 기능 제공
- Connection Pool
    - 지속적인 연결 생성/해제를 가능하게 해 줌
    - 많은 사람들이 함께 사용하는 프로그램일 수록 중요

- 기본 설정

```xml
<?xml version="1.0" encoding="UTF-8"?>
<!DOCTYPE configuration
        PUBLIC "-//mybatis.org/DTD Config 3.0//EN"
        "http://mybatis.org/dtd/mybatis-3-config.dtd" >
```

- MyBatis 실행 설정

```xml
    <settings>
        <setting name="jdbcTypeForNull" value="NULL"/>
    </settings>
    
```

- DB 연결 정보

```xml
<environments default="development">
        <environment id="development">
            <!-- transactionManager를 JDBC로 설정하면 수동으로 transaction을 설정한다는 의미이며
            commit(), rollback()을 코드에서 제어 => JDBC

            만약 MANAGED로 설정하면 자동으로 transaction을 제어하게 할 수 있음 -->
            <transactionManager type="JDBC"/>

            <!-- connection pool을 사용할지 여부 설정 -->
            <dataSource type="POOLED">
                <property name="driver" value="${db.driver}"/>
                <property name="url" value="${db.url}"/>
                <property name="username" value="${db.username}"/>
                <property name="password" value="${db.password}"/>

            </dataSource>
        </environment>
    </environments>
```

- mapper

```xml
<?xml version="1.0" encoding="UTF-8"?>
<!DOCTYPE mapper PUBLIC
        "-//mybatis.org/DTD Mapper 3.0//EN"
        "http://mybatis.org/dtd/mybatis-3-mapper.dtd" >
<!--namespace는 여러 공간을 구분하는 역할-->
<!--보통 table 하나 당 mapper 하나-->
<!--패키지가 클래스 구분해주듯, mapper도 고유한 이름으로 구분-->
<mapper namespace="example.MyBook">
    <!-- parameterType은 파라미터의 데이터 타입을 명시하여 어떻게 보낼지 결정 -->
    <!-- resultType은 SQL 실행 후 받는 데이터를 어떤 데이터타입으로 받아올지 설정 -->
    <select id="selectBYISBNHashMap" parameterType="String"
            resultType="HashMap">
        <!-- SQL 쿼리를 이렇게 작성 -->
        <!-- #{} 안에 파라미터 이름을 통해 값을 가져옴, single quotation('')이 붙음 -->
        <!-- ${} 쓰면 안 됨! 따옴표가 생략된 값이 전달 -->
        <!-- CDATA를 통해서 일반 텍스트 문장임을 명시 -->
        <!-- CDATA는 SQL Query 안의 특수문자(꺽쇠 같은)가 tag로 인식되지 않도록 처리 -->
        <![CDATA[
            SELECT BISBN, BTITLE, BPRICE, BAUTHOR
            FROM BOOK
            WHERE BISBN = #{bisbn}
        ]]>
    </select>
</mapper>
```

- MyBatis의 핵심 객체로, 아래 객체를 통해 SqlSession 객체를 생성, 해당 객체를 통해 SQL 실행이 가능해짐

```java
package example.mybatis;

import org.apache.ibatis.io.Resources;
import org.apache.ibatis.session.SqlSessionFactory;
import org.apache.ibatis.session.SqlSessionFactoryBuilder;

import java.io.Reader;

// 이 객체를 통해 SqlSession 객체를 뽑아내는 것이며, 이 객체를 이용해야 SQL을 실행할 수 있게 됨
public class MyBatisSessionFactory {
    private static SqlSessionFactory sqlSessionFactory;

    static {
        try {
            String resource = "./SqlMapConfig.xml";
            Reader reader = Resources.getResourceAsReader(resource);

            if (sqlSessionFactory == null) {
                sqlSessionFactory = new SqlSessionFactoryBuilder().build(reader);
            }
        } catch (Exception e) {
            e.printStackTrace();
        }
    }

    public static SqlSessionFactory getSqlSessionFactory() {
        return sqlSessionFactory;
    }
}

```

- DAO
- Mapper를 호출하여 비즈니스 로직에서 사용할 데이터를 가져옴

```java
package example.dao;

import org.apache.ibatis.session.SqlSession;
import org.apache.ibatis.session.SqlSessionFactory;

import java.util.HashMap;

public class BookDAO {
    private SqlSessionFactory sqlSessionFactory;

    public BookDAO() {}
    public BookDAO(SqlSessionFactory sqlSessionFactory) {
        this.sqlSessionFactory = sqlSessionFactory;
    }

    //1. bisbn을 이용해서 책 1권의 정보 가져오기
    public HashMap<String, Object> selectByISBNHashMap(String bisbn) {
        HashMap<String, Object> map = null;
        SqlSession sqlSession = sqlSessionFactory.openSession();
        try {
            map = sqlSession.selectOne("example.MyBook.selectBYISBNHashMap", bisbn);
        } catch (Exception e) {
            e.printStackTrace();
        } finally {
            sqlSession.close();
        }
        return map;
    }
}
```

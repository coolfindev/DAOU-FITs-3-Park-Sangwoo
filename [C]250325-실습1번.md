```c
        case 2:
            for (int i = 0; i < transaction_count; i++) {
                printf("ID: %d, 이름: %s, 종목명: %s, 유형: %s, 수량: %d, 가격: %.2f\n",
                    transactions[i].id, transactions[i].name, transactions[i].stock_name,

            }

        case 4:

            printf("잘못된 선택입니다.\n");
        }
    }
    return 0;
```

```c
#define _CRT_SECURE_NO_WARNINGS
#include "myheader.h"
#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <ctype.h>

int transaction_id = 0;

void check_error(OCIError* errhp) {
    text errbuf[512];
    sb4 errcode = 0;
    OCIErrorGet(errhp, 1, NULL, &errcode, errbuf, sizeof(errbuf), OCI_HTYPE_ERROR);
    printf("Oracle Error: %s\n", errbuf);
}

#define MAX_NAME_LENGTH 100
#define MAX_STOCK_NAME_LENGTH 100

void trim_whitespace(char* str) {
    char* end;

    // Leading whitespace
    while (isspace((unsigned char)*str)) str++;

    if (*str == 0)  // All spaces?
        return;

    // Trailing whitespace
    end = str + strlen(str) - 1;
    while (end > str && isspace((unsigned char)*end)) end--;

    // Write new null terminator
    *(end + 1) = 0;
}

void add_transaction(Transcation** transactions, int* transaction_count, OCIEnv* envhp, OCISvcCtx* svchp, OCIError* errhp) {
    Transcation* transaction = (Transcation*)malloc(sizeof(Transcation));
    if (transaction == NULL) {
        printf("메모리 할당에 실패했습니다.\n");
        return;
    }

    printf("고객 이름: ");
    scanf("%s", transaction->name);
    printf("주식 종목명: ");
    scanf("%s", transaction->stock_name);
    printf("거래 유형(0: 매수, 1: 매도): ");
    scanf("%d", &transaction->type);
    printf("거래 수량: ");
    scanf("%d", &transaction->quantity);
    printf("거래 가격: ");
    scanf("%lf", &transaction->price);

    transaction->id = transaction_id++;
    Transcation* temp = (Transcation*)realloc(*transactions, sizeof(Transcation) * (*transaction_count + 1));
    if (temp == NULL) {
        printf("메모리 재할당에 실패했습니다.\n");
        free(transaction);
        return;
    }
    *transactions = temp;
    (*transactions)[*transaction_count] = *transaction;
    (*transaction_count)++;
    free(transaction);
}

void search_transaction(OCIEnv* envhp, OCISvcCtx* svchp, OCIError* errhp) {
    int id;
    printf("검색할 거래 ID: ");
    scanf("%d", &id);

    OCIStmt* stmthp;
    OCIDefine* def1 = NULL, * def2 = NULL, * def3 = NULL, * def4 = NULL, * def5 = NULL, * def6 = NULL;
    char* select_sql = "SELECT ID, NAME, STOCK_NAME, TYPE, QUANTITY, PRICE FROM transactions WHERE ID = :1";
    OCIHandleAlloc(envhp, (void**)&stmthp, OCI_HTYPE_STMT, 0, NULL);
    OCIStmtPrepare(stmthp, errhp, (text*)select_sql, (ub4)strlen(select_sql), OCI_NTV_SYNTAX, OCI_DEFAULT);
    OCIBind* bnd1 = NULL;
    OCIBindByPos(stmthp, &bnd1, errhp, 1, &id, sizeof(id), SQLT_INT, NULL, NULL, NULL, 0, NULL, OCI_DEFAULT);

    if (OCIStmtExecute(svchp, stmthp, errhp, 0, 0, NULL, NULL, OCI_DEFAULT) != OCI_SUCCESS) {
        check_error(errhp);
        OCIHandleFree(stmthp, OCI_HTYPE_STMT);
        return;
    }

    int type, quantity;
    double price;
    char name[100], stock_name[100];
    OCIDefineByPos(stmthp, &def1, errhp, 1, &id, sizeof(id), SQLT_INT, NULL, NULL, NULL, OCI_DEFAULT);
    OCIDefineByPos(stmthp, &def2, errhp, 2, name, sizeof(name), SQLT_CHR, NULL, NULL, NULL, OCI_DEFAULT);
    OCIDefineByPos(stmthp, &def3, errhp, 3, stock_name, sizeof(stock_name), SQLT_CHR, NULL, NULL, NULL, OCI_DEFAULT);
    OCIDefineByPos(stmthp, &def4, errhp, 4, &type, sizeof(type), SQLT_INT, NULL, NULL, NULL, OCI_DEFAULT);
    OCIDefineByPos(stmthp, &def5, errhp, 5, &quantity, sizeof(quantity), SQLT_INT, NULL, NULL, NULL, OCI_DEFAULT);
    OCIDefineByPos(stmthp, &def6, errhp, 6, &price, sizeof(price), SQLT_FLT, NULL, NULL, NULL, OCI_DEFAULT);

    printf("검색 결과:\n");
    printf("--------------------------------------------------\n");
    printf("| ID | NAME | STOCK_NAME | TYPE | QUANTITY | PRICE |\n");
    printf("--------------------------------------------------\n");

    while (OCIStmtFetch2(stmthp, errhp, 1, OCI_DEFAULT, 0, OCI_DEFAULT) == OCI_SUCCESS || OCIStmtFetch2(stmthp, errhp, 1, OCI_DEFAULT, 0, OCI_DEFAULT) == OCI_SUCCESS_WITH_INFO) {
        printf("| %2d | %-20s | %-20s | %4d | %8d | %10.2f |\n", id, name, stock_name, type, quantity, price);
    }

    printf("--------------------------------------------------\n");

    OCIHandleFree(stmthp, OCI_HTYPE_STMT);
}
void modify_transaction(Transcation* transactions, int transaction_count, OCIEnv* envhp, OCISvcCtx* svchp, OCIError* errhp) {
    int id;
    printf("수정할 거래 ID: ");
    scanf("%d", &id);

    int quantity;
    double price;
    printf("새 거래 수량: ");
    scanf("%d", &quantity);
    printf("새 거래 가격: ");
    scanf("%lf", &price);

    for (int i = 0; i < transaction_count; i++) {
        if (transactions[i].id == id) {
            transactions[i].quantity = quantity;
            transactions[i].price = price;
            printf("거래 수정 완료!\n");
            return;
        }
    }
    printf("거래 ID를 찾을 수 없습니다.\n");
}

void delete_transaction(Transcation** transactions, int* transaction_count, OCIEnv* envhp, OCISvcCtx* svchp, OCIError* errhp) {
    int id;
    printf("삭제할 거래 ID: ");
    scanf("%d", &id);

    for (int i = 0; i < *transaction_count; i++) {
        if ((*transactions)[i].id == id) {
            for (int j = i; j < *transaction_count - 1; j++) {
                (*transactions)[j] = (*transactions)[j + 1];
            }
            (*transaction_count)--;
            *transactions = (Transcation*)realloc(*transactions, sizeof(Transcation) * (*transaction_count));
            printf("거래 삭제 완료!\n");
            return;
        }
    }
    printf("거래 ID를 찾을 수 없습니다.\n");
}

void save_transactions_to_file(Transcation* transactions, int transaction_count, const char* filename) {
    FILE* file = fopen(filename, "wb");
    if (file == NULL) {
        perror("파일 열기 실패");
        return;
    }
    fwrite(transactions, sizeof(Transcation), transaction_count, file);
    fclose(file);
}

void save_transactions_to_csv(Transcation* transactions, int transaction_count, const char* filename) {
    FILE* file = fopen(filename, "w");
    if (file == NULL) {
        perror("파일 열기 실패");
        return;
    }
    fprintf(file, "ID,Name,Stock Name,Type,Quantity,Price\n");
    for (int i = 0; i < transaction_count; i++) {
        fprintf(file, "%d,%s,%s,%d,%d,%.2f\n", transactions[i].id, transactions[i].name, transactions[i].stock_name, transactions[i].type, transactions[i].quantity, transactions[i].price);
    }
    fclose(file);
}

void save_transactions_to_db(OCIEnv* envhp, OCISvcCtx* svchp, OCIError* errhp, Transcation* transactions, int transaction_count) {
    OCIStmt* stmthp;
    OCIBind* bnd1 = NULL, * bnd2 = NULL, * bnd3 = NULL, * bnd4 = NULL, * bnd5 = NULL, * bnd6 = NULL;
    char* delete_sql = "DELETE FROM transactions";
    OCIHandleAlloc(envhp, (void**)&stmthp, OCI_HTYPE_STMT, 0, NULL);
    OCIStmtPrepare(stmthp, errhp, (text*)delete_sql, (ub4)strlen(delete_sql), OCI_NTV_SYNTAX, OCI_DEFAULT);

    if (OCIStmtExecute(svchp, stmthp, errhp, 1, 0, NULL, NULL, OCI_COMMIT_ON_SUCCESS) != OCI_SUCCESS) {
        check_error(errhp);
        OCIHandleFree(stmthp, OCI_HTYPE_STMT);
        return;
    }

    OCIHandleFree(stmthp, OCI_HTYPE_STMT);

    for (int i = 0; i < transaction_count; i++) {
        trim_whitespace(transactions[i].name);
        trim_whitespace(transactions[i].stock_name);

        char* insert_sql = "INSERT INTO transactions (ID, NAME, STOCK_NAME, TYPE, QUANTITY, PRICE) VALUES (:1, :2, :3, :4, :5, :6)";
        OCIHandleAlloc(envhp, (void**)&stmthp, OCI_HTYPE_STMT, 0, NULL);
        OCIStmtPrepare(stmthp, errhp, (text*)insert_sql, (ub4)strlen(insert_sql), OCI_NTV_SYNTAX, OCI_DEFAULT);

        OCIBindByPos(stmthp, &bnd1, errhp, 1, &transactions[i].id, sizeof(transactions[i].id), SQLT_INT, NULL, NULL, NULL, 0, NULL, OCI_DEFAULT);
        OCIBindByPos(stmthp, &bnd2, errhp, 2, transactions[i].name, sizeof(transactions[i].name), SQLT_CHR, NULL, NULL, NULL, 0, NULL, OCI_DEFAULT);
        OCIBindByPos(stmthp, &bnd3, errhp, 3, transactions[i].stock_name, sizeof(transactions[i].stock_name), SQLT_CHR, NULL, NULL, NULL, 0, NULL, OCI_DEFAULT);
        OCIBindByPos(stmthp, &bnd4, errhp, 4, &transactions[i].type, sizeof(transactions[i].type), SQLT_INT, NULL, NULL, NULL, 0, NULL, OCI_DEFAULT);
        OCIBindByPos(stmthp, &bnd5, errhp, 5, &transactions[i].quantity, sizeof(transactions[i].quantity), SQLT_INT, NULL, NULL, NULL, 0, NULL, OCI_DEFAULT);
        OCIBindByPos(stmthp, &bnd6, errhp, 6, &transactions[i].price, sizeof(transactions[i].price), SQLT_FLT, NULL, NULL, NULL, 0, NULL, OCI_DEFAULT);

        if (OCIStmtExecute(svchp, stmthp, errhp, 1, 0, NULL, NULL, OCI_COMMIT_ON_SUCCESS) != OCI_SUCCESS) {
            check_error(errhp);
        }
        else {
            printf("데이터베이스에 거래 저장 완료!\n");
        }

        OCIHandleFree(stmthp, OCI_HTYPE_STMT);
    }
}



void load_transactions_from_db(OCIEnv* envhp, OCISvcCtx* svchp, OCIError* errhp, Transcation** transactions, int* transaction_count) {
    OCIStmt* stmthp;
    OCIDefine* def1 = NULL, * def2 = NULL, * def3 = NULL, * def4 = NULL, * def5 = NULL, * def6 = NULL;
    char* select_sql = "SELECT ID, NAME, STOCK_NAME, TYPE, QUANTITY, PRICE FROM transactions";
    OCIHandleAlloc(envhp, (void**)&stmthp, OCI_HTYPE_STMT, 0, NULL);
    OCIStmtPrepare(stmthp, errhp, (text*)select_sql, strlen(select_sql), OCI_NTV_SYNTAX, OCI_DEFAULT);

    if (OCIStmtExecute(svchp, stmthp, errhp, 0, 0, NULL, NULL, OCI_DEFAULT) != OCI_SUCCESS) {
        check_error(errhp);
    }
    else {
        printf("데이터 로드 성공!\n");
    }

    int id, type, quantity;
    double price;
    char name[100], stock_name[100];
    OCIDefineByPos(stmthp, &def1, errhp, 1, &id, sizeof(id), SQLT_INT, NULL, NULL, NULL, OCI_DEFAULT);
    OCIDefineByPos(stmthp, &def2, errhp, 2, name, sizeof(name), SQLT_CHR, NULL, NULL, NULL, OCI_DEFAULT);
    OCIDefineByPos(stmthp, &def3, errhp, 3, stock_name, sizeof(stock_name), SQLT_CHR, NULL, NULL, NULL, OCI_DEFAULT);
    OCIDefineByPos(stmthp, &def4, errhp, 4, &type, sizeof(type), SQLT_INT, NULL, NULL, NULL, OCI_DEFAULT);
    OCIDefineByPos(stmthp, &def5, errhp, 5, &quantity, sizeof(quantity), SQLT_INT, NULL, NULL, NULL, OCI_DEFAULT);
    OCIDefineByPos(stmthp, &def6, errhp, 6, &price, sizeof(price), SQLT_FLT, NULL, NULL, NULL, OCI_DEFAULT);

    sword status;
    *transaction_count = 0;
    while ((status = OCIStmtFetch2(stmthp, errhp, 1, OCI_DEFAULT, 0, OCI_DEFAULT)) == OCI_SUCCESS || status == OCI_SUCCESS_WITH_INFO) {
        Transcation transaction;
        transaction.id = id;
        strcpy(transaction.name, name);
        strcpy(transaction.stock_name, stock_name);
        transaction.type = type;
        transaction.quantity = quantity;
        transaction.price = price;

        Transcation* temp = (Transcation*)realloc(*transactions, sizeof(Transcation) * (*transaction_count + 1));
        if (temp == NULL) {
            printf("메모리 재할당에 실패했습니다.\n");
            break;
        }
        *transactions = temp;
        (*transactions)[*transaction_count] = transaction;
        (*transaction_count)++;
    }

    if (status != OCI_SUCCESS && status != OCI_NO_DATA) {
        check_error(errhp);
    }
    else {
        printf("데이터 로드 완료!\n");
    }

    OCIHandleFree(stmthp, OCI_HTYPE_STMT);
}
```

}function ajax(targetDate) {
  $.ajax({
    async: true,
    type: "GET",
    dataType: "json",
    data: {
      key: "803ad3a4b01e7054913218b932033b33",
      targetDt: targetDate
    },
    url: "http://www.kobis.or.kr/kobisopenapi/webservice/rest/boxoffice/searchDailyBoxOfficeList.json",
    success: function (data) {
      let tbody = $("tbody");
      tbody.empty(); // 기존 데이터 초기화

      let boxOfficeList = data.boxOfficeResult.dailyBoxOfficeList;

      console.log(boxOfficeList);
      boxOfficeList.sort((a, b) => parseInt(a.rank) - parseInt(b.rank));

      // 각 영화에 대해 포스터를 비동기적으로 가져오는 작업
      let moviePromises = boxOfficeList.map(function (movie) {
        return new Promise(function (resolve) {
          getMoviePoster(movie.movieNm, function (posterUrl) {
            // 포스터를 가져온 후에 각 행을 추가하는 작업
            let row = `
              <tr>
                <td>${movie.rank}</td>
                <td><img src="${posterUrl}" alt="영화 포스터" width="50"></td>
                <td>${movie.movieNm}</td>
                <td>${parseInt(movie.audiAcc).toLocaleString()}</td>
                <td>${movie.openDt}</td>
                <td>
                  <button class="btn btn-danger btn-sm" onclick="deleteRow(this)">삭제</button>
                </td>
              </tr>
            `;
            resolve(row);
          });
        });
      });

      // 모든 영화의 포스터가 로드되면 테이블에 추가
      Promise.all(moviePromises).then(function (rows) {
        // 테이블에 순서대로 추가
        tbody.append(rows.join(""));
      });
    },
    error: function (xhr, status, error) {
      console.log(xhr.responseText);
    }
  });
}

function deleteRow(button) {
  $(button).closest("tr").remove();
}

// 날짜 변경 이벤트
$(document).ready(function () {
  let dateInput = $("input[type='date']");

  // 기본값을 오늘 날짜로 설정
  let today = new Date().toISOString().split("T")[0];
  dateInput.val(today);

  // 초기 데이터 불러오기
  ajax(today.replace(/-/g, "")); // YYYYMMDD 형식으로 변환

  // 날짜 변경 시 데이터 다시 불러오기
  dateInput.change(function () {
    let selectedDate = $(this).val().replace(/-/g, ""); // YYYYMMDD 형식으로 변환
    ajax(selectedDate);
  });
});

const KAKAO_API_KEY = "e4a69af3fe5923c9998ec6fc867269cd"; // 여기에 발급받은 키 입력

function getMoviePoster(movieName, callback) {
  $.ajax({
    url: "https://dapi.kakao.com/v2/search/image",
    type: "GET",
    data: { query: movieName, size: 1 },
    headers: { Authorization: `KakaoAK ${KAKAO_API_KEY}` },
    success: function (response) {
      if (response.documents.length > 0) {
        callback(response.documents[0].image_url);
      } else {
        callback("https://via.placeholder.com/50"); // 기본 이미지
      }
    },
    error: function () {
      callback("https://via.placeholder.com/50"); // 에러 발생 시 기본 이미지
    }
  });
}

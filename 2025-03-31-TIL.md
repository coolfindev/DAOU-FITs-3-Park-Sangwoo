import {
  Table,
  TableBody,
  TableCaption,
  TableCell,
  TableHead,
  TableHeader,
  TableRow,
} from "@/components/ui/table";
import { Fragment, useState } from "react";

interface TradeType {
  tradeNumber: number;
  charge: number;
  offerNumber: number;
  exchangeActual: number;
  quantity: number;
  price: number;
  createdAt: string;
  status: string;
  comparison: string;
}
const TradeTable = ({ trades }: { trades: TradeType[] }) => {
  const [expandedRow, setExpandedRow] = useState<number | null>(null);

  const toggleRow = (index: number) => {
    setExpandedRow(expandedRow === index ? null : index);
  };

  return (
    <Table className="text-xs">
      <TableCaption>
        2025년 03월 01일 ~ 2025년 03월 31일의 주문 내역입니다.
      </TableCaption>
      <TableHeader>
        <TableRow>
          <TableHead>일자</TableHead>
          <TableHead>구분</TableHead>
          <TableHead>거래소</TableHead>
          <TableHead>종목명</TableHead>
          <TableHead className="text-right">수량</TableHead>
          <TableHead className="text-right">단가</TableHead>
          <TableHead className="text-right">거래금액</TableHead>
        </TableRow>
      </TableHeader>
      <TableBody>
        {trades.length ? (
          trades.map((trade: TradeType, index) => (
            <Fragment key={trade.createdAt + index}>
              {/* 기본 행 */}
              <TableRow
                onClick={() => toggleRow(index)}
                className="cursor-pointer"
              >
                <TableCell>{trade.createdAt.split(" ")[0]}</TableCell>
                <TableCell>{trade.status}</TableCell>
                <TableCell>{trade.exchangeActual ? "NXT" : "KRX"}</TableCell>
                <TableCell>{"삼성전자"}</TableCell>
                <TableCell className="text-right">{trade.quantity}</TableCell>
                <TableCell className="text-right">
                  {trade.price.toLocaleString()}
                </TableCell>
                <TableCell className="text-right">
                  {(trade.price * trade.quantity).toLocaleString()}
                </TableCell>
              </TableRow>

              {/* 확장되는 상세 정보 */}
              {expandedRow === index && (
                <TableRow className="bg-gray-100">
                  <TableCell colSpan={7} className="p-4 bg-slate-50">
                    <div className="space-y-2 flex flex-col">
                      <h3 className="text-lg font-semibold text-gray-800">
                        거래 상세 정보
                      </h3>
                      <div className="flex flex-col gap-y-2">
                        <div className="flex gap-x-4">
                          <p className="font-medium w-[100px]">거래 ID</p>
                          <p className="font-light">{index + 1}</p>
                        </div>
                        <div className="flex gap-x-4">
                          <p className="font-medium w-[100px]">거래 시간</p>
                          <p className="font-light">
                            {trade.createdAt
                              .split(" ")[1]
                              .split(".")[0]
                              .split(":")
                              .map((unit, i) =>
                                i === 0
                                  ? `${unit}시`
                                  : i === 1
                                  ? `${unit}분`
                                  : `${unit}초`
                              )
                              .join(" ")}
                          </p>
                        </div>
                        <div className="flex gap-x-4">
                          <p className="font-medium w-[100px]">거래 금액</p>
                          <p className="font-light">
                            {(trade.price * trade.quantity).toLocaleString()} 원
                          </p>
                        </div>
                        <div className="flex gap-x-4">
                          <p className="font-medium w-[100px]">수수료</p>
                          <p className="font-light">
                            {trade.charge.toLocaleString()} 원
                          </p>
                        </div>
                        <div className="flex gap-x-4">
                          <p className="font-medium w-[100px]">거래소</p>
                          <p className="font-light">
                            {trade.exchangeActual
                              ? "대체거래소(ATS)"
                              : "한국거래소(KRX)"}
                          </p>
                        </div>
                        <div className="flex gap-x-4">
                          <p className="font-medium w-[100px]">
                            반대 거래소 대비
                          </p>
                          <p className="font-light">{trade?.comparison} 원</p>
                        </div>
                      </div>
                    </div>
                  </TableCell>
                </TableRow>
              )}
            </Fragment>
          ))
        ) : (
          <TableRow>
            <TableCell colSpan={7} className="text-center">
              데이터가 존재하지 않습니다.
            </TableCell>
          </TableRow>
        )}
      </TableBody>
    </Table>
  );
};

export default TradeTable;

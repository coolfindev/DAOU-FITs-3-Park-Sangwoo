package first;

import java.sql.Connection;
import java.sql.DriverManager;
import java.sql.PreparedStatement;
import java.sql.SQLException;

public class TransactionTest {
    public static void main(String[] args) {
        try {
            //드라이버 설정
            Class.forName("oracle.jdbc.driver.OracleDriver");

            //dbms 연결
            Connection conn = null;
            String url = "jdbc:oracle:thin:@localhost:1521:xe";
            String user = "C##DEV";
            String password = "1423";

            //statement 생성
            conn = DriverManager.getConnection(url, user, password);
            //transaction을 jdbc를 이용해서 설정하기 위해서 Connection에서 수행
            //아래 구문 없이 진행되었을 시, 정상적으로 종료(close)되면 conn.commit()으로 간주함
            conn.setAutoCommit(false);
            //preparedStatement를 사용하기 위해 ?, ?로 values 지정
            String sql = "insert into membertbl values(?,?)";
            PreparedStatement ps = conn.prepareStatement(sql);
            ps.setString(1, "1");
            ps.setString(2, "홍길동");

            //쿼리 실행
            int res = ps.executeUpdate();

            //결과 확인
            if(res == 1) {
                System.out.println("처리완료");
            }

            //transaction 저장 및 종료
            conn.commit();

            //연결 해제
            conn.close();
            ps.close();
        } catch (ClassNotFoundException e) {
            System.out.println("클래스가 없어요");
        } catch (SQLException e) {
            System.out.println("SQL 이슈");
        }
    }
}

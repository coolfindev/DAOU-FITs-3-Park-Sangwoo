- 상태관리 과정에서 useQuery <-> useMutation 간 데이터 전달에 관한 고민(어떻게 해야 개선되는지 더욱 고민해봐야 함)


```js
'use client'
import publicAPI from '@/api/publicAPI'
import BaseContainer from '@/components/BaseContainer'
import GlobalHeader from '@/components/GlobalHeader'
import ServiceApply from '@/components/ServiceApply'
import ServiceModal from '@/components/ServiceModal'
import { useMutation, useQuery, useQueryClient } from '@tanstack/react-query'
import React, { useEffect, useState } from 'react'
import toast from 'react-hot-toast'

import UpperICON from '@/static/icons/upper-icon.png'
import LowerICON from '@/static/icons/lower-icon.png'
import Image from 'next/image'
import TickChart from '@/components/chart'

interface StockType {
  stockId: number
  stockCode: string
  stockName: string
  stockMarket: string
  recentClosePrice: number
  createdAt: string
  updatedAt: string
  deletedAt: string
}
interface ChartStockType {
  stockCode: string
  stockId: number
  stockName: string
  chartData: ChartType[]
}

interface ChartType {
  date: string
  open: number
  close: number
  high: number
  low: number
  change: number
  changeRate: number
}

function Service() {
  const [modal, setModal] = useState<StockType>()
  const [searchedStockList, setSearchedStockList] = useState<StockType[]>([])

  const closeModal = () => {
    setModal(undefined)
  }

  const openModal = (stock: StockType) => {
    setModal(stock)
  }

  //잔액 조회
  const { data: accountBalance } = useQuery({
    queryKey: ['accountBalnce'],
    queryFn: async () => {
      const response = await publicAPI.get('/api/account/balance', {
        params: {
          accountId: sessionStorage.getItem('accountid'),
        },
      })
      return response.data
    },
  })

  // 서비스 신청 여부
  const { data: isService } = useQuery({
    queryKey: ['serviceApply'],
    queryFn: async () => {
      const response = await publicAPI.get('/api/account/contract', {
        params: {
          accountId: sessionStorage.getItem('accountid'),
          contractId: 1,
        },
      })
      return response.data
    },
  })

  // 신청한 서비스 정보
  const { data: serviceDetail, isSuccess: isServiceDetailLoaded } = useQuery({
    queryKey: ['serviceDetail'],
    queryFn: async () => {
      const response = await publicAPI.get('/api/account/auto', {
        params: {
          accountId: sessionStorage.getItem('accountid'),
          contractId: 1,
        },
      })

      return response.data
    },
    refetchOnMount: 'always',
    //캐시 없이
    staleTime: 0,
  })

  useEffect(() => {
    console.log(isServiceDetailLoaded)
    if (isServiceDetailLoaded) {
      getServiceDetail()
    }
    // eslint-disable-next-line react-hooks/exhaustive-deps
  }, [isServiceDetailLoaded, serviceDetail?.stockId])

  //신청한 서비스 종목 정보
  const [serviceStock, setServiceStock] = useState<ChartStockType>()
  const { mutate: getServiceDetail } = useMutation({
    mutationKey: ['stockDetail'],
    mutationFn: async () => {
      const response = await publicAPI.get('/api/stock/chart', {
        params: {
          stockId: serviceDetail?.stockId,
          startDate: '2025-02-01',
          endDate: '2025-02-25',
        },
      })
      return response.data
    },

    onSuccess: (data: ChartStockType) => {
      setServiceStock(data)
    },
  })

  const { data: stockList } = useQuery({
    queryKey: ['stockList'],
    queryFn: async () => {
      const response = await publicAPI.get('/api/stock/all')

      if (response.status === 200) {
        setSearchedStockList(response.data.stocks)
      }
      return response.data.stocks
    },
  })

  const handleSearch = (e: React.ChangeEvent<HTMLInputElement>) => {
    const value = e.target.value

    if (!value) {
      setSearchedStockList(stockList)
      return
    }

    const filteredStockList = stockList.filter((stock: StockType) => {
      return stock.stockName.includes(value)
    })

    setSearchedStockList(filteredStockList)
  }

  const cancelService = () => {
    deleteService()
  }

  const queryClient = useQueryClient()
  const { mutate: deleteService } = useMutation({
    mutationFn: async () => {
      const response = await publicAPI.delete('/api/account/auto', {
        params: {
          accountId: sessionStorage.getItem('accountid'),
          contractId: 1,
        },
      })
      return response.data
    },

    onSuccess: () => {
      toast.success('자동주문이 취소되었습니다.')

      queryClient.invalidateQueries({
        queryKey: ['serviceApply'],
      })

      queryClient.invalidateQueries({
        queryKey: ['serviceDetail'],
      })

      queryClient.invalidateQueries({
        queryKey: ['stockList'],
      })

      queryClient.invalidateQueries({
        queryKey: ['accountBalnce'],
      })

      queryClient.invalidateQueries({
        queryKey: ['stockDetail'],
      })

      queryClient.invalidateQueries({
        queryKey: ['stockBalanceList'],
      })

      setServiceStock(undefined)
    },
  })

  return (
    <BaseContainer>
      <GlobalHeader text="주식적립 자동주문" />

      {isService ? (
        <div className="flex flex-col items-center justify-start gap-y-10 py-10 focus:outline-none">
          <div className="flex w-full flex-col items-center px-8 py-4">
            <div className="flex w-full flex-col">
              <div className="flex items-center justify-between">
                <h3 className="w-full text-2xl font-bold">
                  {serviceStock?.stockName}
                </h3>
                {accountBalance.balance <
                  Number(
                    serviceStock?.chartData[serviceStock?.chartData.length - 1]
                      ?.close,
                  ) && (
                  <span className="shrink-0 rounded-sm border border-red-300 px-2 py-1 text-sm text-red-400">
                    잔액 부족
                  </span>
                )}
              </div>

              <span className="text-sm text-gray-400">
                KOSPI {serviceStock?.stockCode}
              </span>
            </div>

            {/* 차트 마지막 행 데이터 */}
            <div
              className={`flex w-full flex-col gap-y-2 ${
                Number(
                  serviceStock?.chartData[serviceStock?.chartData.length - 1]
                    ?.close,
                ) -
                  Number(
                    serviceStock?.chartData[serviceStock?.chartData.length - 2]
                      ?.close,
                  ) >
                0
                  ? 'text-[#eb058c]'
                  : 'text-[#013976]'
              }`}
            >
              <span className="flex w-full items-center justify-center text-4xl font-bold">
                {serviceStock?.chartData[
                  serviceStock?.chartData.length - 1
                ]?.close.toLocaleString('ko-kr')}
              </span>

              <div className="flex w-full items-center justify-center gap-x-4 text-sm">
                <div className="flex items-center gap-x-1">
                  {Number(
                    serviceStock?.chartData[serviceStock?.chartData.length - 1]
                      ?.close,
                  ) -
                    Number(
                      serviceStock?.chartData[
                        serviceStock?.chartData.length - 2
                      ]?.close,
                    ) >
                  0 ? (
                    <Image
                      src={UpperICON}
                      alt="UP"
                      className="h-4 w-4"
                    />
                  ) : (
                    <Image
                      src={LowerICON}
                      alt="DOWN"
                      className="h-4 w-4"
                    />
                  )}
                  {(
                    Number(
                      serviceStock?.chartData[
                        serviceStock?.chartData.length - 1
                      ]?.close,
                    ) -
                    Number(
                      serviceStock?.chartData[
                        serviceStock?.chartData.length - 2
                      ]?.close,
                    )
                  ).toLocaleString('ko-kr')}
                </div>

                {/* 위 값의 백분율 */}
                <span>
                  {(
                    ((Number(
                      serviceStock?.chartData[
                        serviceStock?.chartData.length - 1
                      ]?.close,
                    ) -
                      Number(
                        serviceStock?.chartData[
                          serviceStock?.chartData.length - 2
                        ]?.close,
                      )) /
                      Number(
                        serviceStock?.chartData[
                          serviceStock?.chartData.length - 2
                        ]?.close,
                      )) *
                    100
                  ).toFixed(2) + '%'}
                </span>
              </div>
            </div>
          </div>
          <hr className="h-2 w-full bg-gray-50" />

          {/* 기본정보 */}
          <div className="flex w-full flex-col items-center justify-center gap-y-4 border-gray-200">
            <h3 className="w-full px-8 font-bold">종목 정보</h3>
            {/* 종목정보 warpper */}
            <div className="flex w-full flex-col border-y">
              <div className="flex w-full items-center justify-between border-b py-4">
                <span className="flex-1 text-center text-gray-400">종목명</span>
                <span className="flex-1 text-center">
                  {serviceDetail && serviceStock?.stockName}
                </span>
              </div>
              <div className="flex w-full items-center justify-between border-b py-4">
                <span className="flex-1 text-center text-gray-400">
                  서비스 횟수
                </span>
                <span className="flex-1 text-center">
                  {serviceDetail && serviceDetail?.discountCnt}회
                </span>
              </div>
              <div className="flex w-full items-center justify-between border-b py-4">
                <span className="flex-1 text-center text-gray-400">
                  주문 주기
                </span>
                <span className="flex-1 text-center">
                  {serviceDetail && serviceDetail?.cycleTime === 'DAY'
                    ? '매일'
                    : serviceDetail?.cycleTime === 'WEEK'
                      ? '매주'
                      : '매월'}
                </span>
              </div>
              <div className="flex w-full items-center justify-between border-b py-4">
                <span className="flex-1 text-center text-gray-400">금액</span>
                <span className="flex-1 text-center">
                  {serviceDetail &&
                    (serviceDetail?.amount).toLocaleString('ko-kr') + '원'}
                </span>
              </div>
              <div className="flex w-full items-center justify-between py-4">
                <span className="flex-1 text-center text-gray-400">
                  등록일자
                </span>
                <span className="flex-1 text-center">
                  {serviceDetail &&
                    new Date(serviceDetail?.createdAt).toLocaleDateString('en')}
                </span>
              </div>
            </div>
          </div>

          <hr className="h-2 w-full bg-gray-50" />
          {/* 차트 */}
          <div className="flex w-full flex-col px-8">
            {serviceStock && <TickChart chart={serviceStock} />}
          </div>

          <hr className="h-2 w-full bg-gray-50" />
          {/* 취소 */}
          <div className="flex w-full items-center justify-end px-8">
            <button
              onClick={cancelService}
              className="self-end rounded-sm border border-red-300 px-4 py-2 text-sm text-red-400"
            >
              취소
            </button>
          </div>
        </div>
      ) : (
        <div className="flex flex-col items-center justify-start gap-y-10 px-8 py-10 focus:outline-none">
          {/* 검색창 */}
          <input
            type="text"
            className="h-12 w-full rounded-md border-2 border-gray-100 px-4 focus:border-[#013976] focus:outline-none"
            placeholder="종목명 검색"
            onChange={handleSearch}
          />
          {/* 검색결과리스트 */}
          <div className="flex w-full flex-col">
            {searchedStockList
              ? searchedStockList.map((stock: StockType) => (
                  <ServiceApply
                    key={stock.stockName + '_list_item'}
                    stock={stock}
                    openModal={openModal}
                  />
                ))
              : stockList?.map((stock: StockType) => (
                  <ServiceApply
                    key={stock.stockName + '_list_item'}
                    stock={stock}
                    openModal={openModal}
                  />
                ))}
          </div>
        </div>
      )}
      {/* 모달 */}
      {modal && (
        <ServiceModal
          stock={modal}
          closeModal={closeModal}
        />
      )}
    </BaseContainer>
  )
}

export default Service
```

### 발표

**전제**

- 모든 사용자는 일련의 과정을 통해 회원가입을 성공적으로 수행했다
- 모든 사용자는 사전에 일련의 과정을 통해 증거금률 100%의 계좌를 한 개 가지고 있다
- 서비스 시연의 진행을 위해 1일을 10초, 1주일을 20초, 1달을 30초로 한다

**시연 시나리오**

1. (사전 가입)회원 가입
    1. sqldevloper를 통해 db에 깡 가입
    2. 계정 생성 쿼리
    
    ```sql
    INSERT INTO USERS (email, password, name) VALUES ('test', '1234', '박선홍');
    INSERT INTO ACCOUNTS (user_id, account_no, password, margine_rate, balance ) VALUES (2, '0000-1234', '1234', 100, 10000000);
    COMMIT;
    ```
    
2. **로그인**
    1. 이미 생성된 계정을 통해 로그인을 수행
    2. 아이디 찾기, 비밀번호 찾기는 별도로 구현되어있지 않음
    - 계정 리스트
        
        
        | **ID** | **PW** |
        | --- | --- |
        | admin | 1423 |
        | (예정)test | (예정)1234 |
3. **계좌 관리**
    1. 계좌는 전제에 따라 생성이 되어있고, 기본 정보를 메인 화면에서 확인할 수 있음
    2. 입금
        1. 특정 금액을 적게 입금
    3. 출금
        1. 특정 금액을 출금
4. **자동 주문**
    1. 종목에 대한 전체 리스트 확인하고 가격에 대한 소개
    2. `다우기술`종목 검색 후 등록
    3. 국내 주식, 그 중 10개에 대해서만 자료를 수집했음을 강조
    4. 매일(10초) 10만원 제한으로 등록
5. **자동주문 상세 페이지**
    1. 적게 입금했기 때문에 ‘잔액부족’이 발생
    2. 이를 위해 약 100만원 입금
    3. 이후 이체내역에서 돈이 종목구매로 빠지는 것을 확인
6. **종목 페이지**
    1. 방금 10만원 아래에서 해당하는 약 N주를 매수했기 때문에 아래 이렇게 나옴  
    2. 말씀드리는 동안 N초가 지났기에 이렇게 새로고침을 한다면 보유종목 양이 늘어난 것을 확인할 수 있음
7. **자동주문 취소, 재등록**
    1. 앞서 진행했던 다우기술 종목을 취소하고 `키움증권`으로 새로 등록함
    2. 10초가 지난 뒤 종목 탭에서 변화량 확인
    3. 이를 통해 서비스 내에서 종목을 취소하고 등록하더라도 보유 종목은 여기서 한 눈에 확인 가능함을 강조
8. **타 계정**
9. 이러한 포트폴리오가 누적됐을 때 좌측(기존 보여주기 용 계정)과 같아짐 → 자산관리 용이

package booksearch;

import booksearch.vo.Book;
import javafx.collections.FXCollections;
import javafx.collections.ObservableList;
import javafx.fxml.FXML;
import javafx.fxml.Initializable;
import javafx.scene.control.*;
import javafx.scene.control.cell.PropertyValueFactory;
import javafx.scene.control.cell.TextFieldTableCell;
import javafx.util.converter.IntegerStringConverter;

import java.net.URL;
import java.sql.*;
import java.util.ResourceBundle;
import java.util.Scanner;

public class JavaFXTableViewController implements Initializable {

    @FXML
    private TableView<Book> tableView;

    @FXML
    private TextField textField;

    @FXML
    private Button searchBtn;

    @FXML private TableColumn<Book, String> isbnCol;
    @FXML private TableColumn<Book, String> titleCol;
    @FXML private TableColumn<Book, String> authorCol;
    @FXML private TableColumn<Book, Integer> priceCol;

    @Override
    public void initialize(URL url, ResourceBundle resourceBundle) {
        isbnCol.setCellValueFactory(new PropertyValueFactory<>("bisbn"));
        titleCol.setCellValueFactory(new PropertyValueFactory<>("btitle"));
        priceCol.setCellValueFactory(new PropertyValueFactory<>("bprice"));
        authorCol.setCellValueFactory(new PropertyValueFactory<>("bauthor"));

        //수정 가능하게 만듦
        isbnCol.setCellFactory(TextFieldTableCell.forTableColumn());
        titleCol.setCellFactory(TextFieldTableCell.forTableColumn());
        authorCol.setCellFactory(TextFieldTableCell.forTableColumn());
        priceCol.setCellFactory(TextFieldTableCell.forTableColumn(new IntegerStringConverter()));
        tableView.setEditable(true);

        //수정 로직
        isbnCol.setOnEditCommit(event -> {
            Book book = event.getRowValue();
            String newIsbn = event.getNewValue();
            updateBookInDB(book.getBisbn(), "bisbn", newIsbn);
            book.setBisbn(newIsbn);
        });

        titleCol.setOnEditCommit(event -> {
            Book book = event.getRowValue();
            String newTitle = event.getNewValue();
            updateBookInDB(book.getBisbn(), "btitle", newTitle);
            book.setBtitle(newTitle);
        });

        authorCol.setOnEditCommit(event -> {
            Book book = event.getRowValue();
            String newAuthor = event.getNewValue();
            updateBookInDB(book.getBisbn(), "bauthor", newAuthor);
            book.setBauthor(newAuthor);
        });

        priceCol.setOnEditCommit(event -> {
            Book book = event.getRowValue();
            Integer newPrice = event.getNewValue();
            updateBookInDB(book.getBisbn(), "bprice", String.valueOf(newPrice));
            book.setBprice(newPrice);
        });

        // TableView에 ContextMenu 추가
        ContextMenu contextMenu = new ContextMenu();
        MenuItem deleteItem = new MenuItem("삭제");

        // 삭제 기능 구현
        deleteItem.setOnAction(event -> {
            Book selectedBook = tableView.getSelectionModel().getSelectedItem();
            if (selectedBook != null) {
                deleteBookFromDB(selectedBook.getBisbn());
                tableView.getItems().remove(selectedBook);
            }
        });

        contextMenu.getItems().add(deleteItem);
        //삭제로직
        //setRowFactory를 통해 메뉴가 front에 생성
        tableView.setRowFactory(tv -> {
            TableRow<Book> row = new TableRow<>();
            row.setOnContextMenuRequested(event -> {
                if (!row.isEmpty()) {
                    contextMenu.show(row, event.getScreenX(), event.getScreenY());
                } else {
                    contextMenu.hide();
                }
            });
            return row;
        });

        try {
            Class.forName("oracle.jdbc.driver.OracleDriver");

            String jdbc_url = "jdbc:oracle:thin:@localhost:1521:xe";
            String user = "C##DEV";
            String password = "1423";
            Connection con = DriverManager.getConnection(jdbc_url, user, password);

            String sql = "select bisbn, btitle, bprice, bauthor from book";
            PreparedStatement ps = con.prepareStatement(sql);
            ResultSet rs = ps.executeQuery();

            ObservableList<Book> books = FXCollections.observableArrayList();
            while (rs.next()) {
                books.add(new Book(rs.getString("bisbn"), rs.getString("btitle"), rs.getInt("bprice"), rs.getString("bauthor")));
            }
            tableView.setItems(books);

            searchBtn.setOnAction(event -> {
                books.clear();

                try {
                    PreparedStatement ps2 = con.prepareStatement(sql+" where btitle LIKE ?");
                    ps2.setString(1, "%"+textField.getText()+"%");

                    ResultSet rs2 = ps2.executeQuery();
                    while (rs2.next()) {
                        books.add(new Book(rs2.getString("bisbn"), rs2.getString("btitle"), rs2.getInt("bprice"), rs2.getString("bauthor")));
                    }

                    tableView.setItems(books);
                } catch (SQLException e) {
                    throw new RuntimeException(e);
                }


                tableView.setItems(books);
            });
        } catch (ClassNotFoundException e) {
            System.out.println("클래스 식별 불가");
        } catch (SQLException e) {
            System.out.println("SQL 접근 불가");
        }
    }

    private void updateBookInDB(String bisbn, String column, String newValue) {
        String sql = "UPDATE book SET " + column + " = ? WHERE bisbn = ?";

        try (Connection con = DriverManager.getConnection(
                "jdbc:oracle:thin:@localhost:1521:xe", "C##DEV", "1423");
             PreparedStatement ps = con.prepareStatement(sql)) {

            if (column.equals("bprice")) {
                ps.setInt(1, Integer.parseInt(newValue)); // 가격은 숫자로 변환
            } else {
                ps.setString(1, newValue);
            }
            ps.setString(2, bisbn); // WHERE 조건에 기존 ISBN 사용

            int affectedRows = ps.executeUpdate();
            if (affectedRows > 0) {
                System.out.println(column + " 업데이트 성공: " + newValue);
            } else {
                System.out.println(column + " 업데이트 실패");
            }

        } catch (SQLException e) {
            e.printStackTrace();
        }
    }

    private void deleteBookFromDB(String bisbn) {
        String sql = "DELETE FROM book WHERE bisbn = ?";

        try (Connection con = DriverManager.getConnection(
                "jdbc:oracle:thin:@localhost:1521:xe", "C##DEV", "1423");
             PreparedStatement ps = con.prepareStatement(sql)) {

            ps.setString(1, bisbn);
            int affectedRows = ps.executeUpdate();

            if (affectedRows > 0) {
                System.out.println("삭제 성공: " + bisbn);
            } else {
                System.out.println("삭제 실패: " + bisbn);
            }

        } catch (SQLException e) {
            e.printStackTrace();
        }
    }

}

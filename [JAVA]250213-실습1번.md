```fxml
<?xml version="1.0" encoding="UTF-8"?>
<!DOCTYPE configuration
        PUBLIC "-//mybatis.org/DTD Config 3.0//EN"
        "http://mybatis.org/dtd/mybatis-3-config.dtd" >
<configuration>
    <!--jdbc driver 클래스와 데이터베이스 연결 설정 -->
    <!--이 연결 설정에 대한 내용은 xml 안에 직접 작성하지 않고 외부 파일에 작성해서 해당 파일을 property로 사용-->

    <properties resource="./driver.properties"/>
    
    <!-- MyBatis 실행 설정 -->
    <settings>
        <setting name="jdbcTypeForNull" value="NULL"/>
    </settings>
    
    <!-- TypeAlias가 나올 수 있음 -->
    <typeAliases>
        <typeAlias type="test.vo.Book" alias="Book" />
    </typeAliases>
    
    <!-- db 연결 정보 -->
    <environments default="development">
        <environment id="development">
            <!-- transactionManager를 JDBC로 설정하면 수동으로 transaction을 설정한다는 의미이며
            commit(), rollback()을 코드에서 제어 => JDBC

            만약 MANAGED로 설정하면 자동으로 transaction을 제어하게 할 수 있음 -->
            <transactionManager type="JDBC"/>

            <!-- connection pool을 사용할지 여부 설정 -->
            <dataSource type="POOLED">
                <property name="driver" value="${db.driver}"/>
                <property name="url" value="${db.url}"/>
                <property name="username" value="${db.username}"/>
                <property name="password" value="${db.password}"/>

            </dataSource>
        </environment>
    </environments>

    <!-- SQL 구문을 작성하는 maper라는 xml파일 경로 지정 -->
    <mappers>
        <mapper resource="./sqlmap/Book.xml"/>
    </mappers>
</configuration>
```

```xml
    <select id="selectByTitleHashMap" parameterType="String"
            resultType="HashMap">
        <!-- SQL 쿼리를 이렇게 작성 -->
        <!-- #{} 안에 파라미터 이름을 통해 값을 가져옴, single quotation('')이 붙음 -->
```

### dao

```java
package test.dao;

import javafx.collections.FXCollections;
import javafx.collections.ObservableList;
import org.apache.ibatis.session.SqlSession;
import org.apache.ibatis.session.SqlSessionFactory;
import test.vo.Book;

import java.sql.*;
import java.util.HashMap;
import java.util.List;

public class BookDAO {
    private SqlSessionFactory sqlSessionFactory;

    public BookDAO() {}
    public BookDAO(SqlSessionFactory sqlSessionFactory) {
        this.sqlSessionFactory = sqlSessionFactory;
    }

    //기본 전체 조회
    public List<HashMap<String, Object>> selectAllHashMap() {
        List<HashMap<String, Object>> map = null;
        SqlSession sqlSession = sqlSessionFactory.openSession();
        try {
            map = sqlSession.selectList("test.TestBook.selectAllHashMap");
        } catch (Exception e) {
            e.printStackTrace();
        } finally {
            sqlSession.close();
        }
        return map;
    }

    //책 제목 키워드 검색 조회
    public List<HashMap<String, Object>> selectByTitleHashMap(String btitle) {
            List<HashMap<String, Object>> map = null;
            SqlSession sqlSession = sqlSessionFactory.openSession();
        try {
            map = sqlSession.selectList("test.TestBook.selectByTitleHashMap", btitle);
        } catch (Exception e) {
            e.printStackTrace();
        } finally {
            sqlSession.close();
        }
        return map;
    }

    // 항목 수정
    public boolean updateBook(String bisbn, String column, Object newValue) {
        SqlSession sqlSession = sqlSessionFactory.openSession();
        boolean success = false;

        try {
            // 업데이트할 데이터 생성
            if ("bprice".equals(column) && newValue instanceof String) {
                // price 값이 String으로 전달된 경우, Integer로 변환
                newValue = Integer.parseInt((String) newValue);
            }

            HashMap<String, Object> paramMap = new HashMap<>();
            paramMap.put("bisbn", bisbn);
            paramMap.put("column", column);
            paramMap.put("newValue", newValue);

            // MyBatis 업데이트 실행
            int affectedRows = sqlSession.update("test.TestBook.updateBook", paramMap);
            sqlSession.commit(); // 변경 사항 반영

            success = (affectedRows > 0);
        } catch (Exception e) {
            e.printStackTrace();
            sqlSession.rollback(); // 오류 발생 시 롤백
        } finally {
            sqlSession.close();
        }

        return success;
    }

    public boolean deleteBook(String bisbn) {
        SqlSession sqlSession = sqlSessionFactory.openSession();
        boolean success = false;
        try {
            int result = sqlSession.delete("test.TestBook.deleteBook", bisbn);

            sqlSession.commit();
            success = (result > 0);
        } catch (Exception e) {
            e.printStackTrace();
        } finally {
            sqlSession.close();
        }
        return success;
    }

    public boolean createBook(String bisbn, String btitle, int bprice, String bauthor) {
        SqlSession sqlSession = sqlSessionFactory.openSession();
        boolean success = false;

        try {

            HashMap<String, Object> paramMap = new HashMap<>();
            paramMap.put("bisbn", bisbn);
            paramMap.put("btitle", btitle);
            paramMap.put("bprice", bprice);
            paramMap.put("bauthor", bauthor);
            int affectedRows = sqlSession.insert("test.TestBook.createBook", paramMap);
            sqlSession.commit();

            success = (affectedRows > 0);
        } catch (Exception e) {
            e.printStackTrace();
        } finally {
            sqlSession.close();
        }

        return success;
    }
}
```

```java
package test.controller;

import javafx.beans.property.SimpleObjectProperty;
import javafx.beans.property.SimpleStringProperty;
import javafx.collections.FXCollections;
import javafx.collections.ObservableList;
import javafx.fxml.FXML;
import javafx.fxml.Initializable;
import javafx.scene.control.*;
import javafx.scene.control.cell.PropertyValueFactory;
import javafx.scene.control.cell.TextFieldTableCell;
import javafx.util.converter.IntegerStringConverter;
import org.apache.ibatis.session.SqlSessionFactory;
import test.dao.BookDAO;
import test.mybatis.MyBatisSessionFactory;
import test.vo.Book;

import java.net.URL;
import java.util.HashMap;
import java.util.List;
import java.util.ResourceBundle;

public class JavaFXTableViewController implements Initializable {

    @FXML private TableView<HashMap<String, Object>> tableView;

    //search
    @FXML private TextField searchField;
    @FXML private Button searchBtn;

    //del
    @FXML private Button deleteBtn;

    //create
    @FXML private TextField isbnField;
    @FXML private TextField titleField;
    @FXML private TextField authorField;
    @FXML private TextField priceField;

    @FXML private TableColumn<HashMap<String, Object>, String> isbnCol;
    @FXML private TableColumn<HashMap<String, Object>, String> titleCol;
    @FXML private TableColumn<HashMap<String, Object>, String> authorCol;
    @FXML private TableColumn<HashMap<String, Object>, Integer> priceCol;

    @FXML private Button createBtn;

    @Override
    public void initialize(URL url, ResourceBundle resourceBundle) {
        setupTableColumns();
        loadBookData();
        setupSearchButton();
        setupRowEditing();
        setupRowDeleting();
        setupCreating();
    }

    private void setupTableColumns() {
        isbnCol.setCellValueFactory(data -> new SimpleStringProperty((String) data.getValue().get("BISBN")));
        titleCol.setCellValueFactory(data -> new SimpleStringProperty((String) data.getValue().get("BTITLE")));
        authorCol.setCellValueFactory(data -> new SimpleStringProperty((String) data.getValue().get("BAUTHOR")));

        // 안전한 변환을 위해 Number 타입 확인 후 int 변환
        priceCol.setCellValueFactory(data -> {
            Object value = data.getValue().get("BPRICE");
            if (value instanceof Number) {
                return new SimpleObjectProperty<>(((Number) value).intValue()); // 변환 후 반환
            }
            return new SimpleObjectProperty<>(0); // 기본값 0
        });

        // 셀 편집 가능하도록 설정
        isbnCol.setCellFactory(TextFieldTableCell.forTableColumn());
        titleCol.setCellFactory(TextFieldTableCell.forTableColumn());
        authorCol.setCellFactory(TextFieldTableCell.forTableColumn());
        priceCol.setCellFactory(TextFieldTableCell.forTableColumn(new IntegerStringConverter()));

        tableView.setEditable(true);
    }

    SqlSessionFactory factory = MyBatisSessionFactory.getSqlSessionFactory();
    BookDAO bookDAO = new BookDAO(factory);

    private void loadBookData() {
        List<HashMap<String, Object>> books = bookDAO.selectAllHashMap();
        ObservableList<HashMap<String, Object>> bookObservableList = FXCollections.observableArrayList(books);
        tableView.setItems(bookObservableList);
    }

    private void setupSearchButton() {
        searchBtn.setOnAction(event -> {
            String keyword = searchField.getText();
            List<HashMap<String,Object>> books = bookDAO.selectByTitleHashMap(keyword);
            ObservableList<HashMap<String, Object>> bookObservableList = FXCollections.observableArrayList(books);
            tableView.setItems(bookObservableList);
        });
    }

    private void setupRowEditing() {
        isbnCol.setOnEditCommit(event -> updateBook(event.getRowValue(), "BISBN", event.getNewValue()));
        titleCol.setOnEditCommit(event -> updateBook(event.getRowValue(), "BTITLE", event.getNewValue()));
        authorCol.setOnEditCommit(event -> updateBook(event.getRowValue(), "BAUTHOR", event.getNewValue()));
        priceCol.setOnEditCommit(event -> updateBook(event.getRowValue(), "BPRICE", String.valueOf(event.getNewValue())));
    }

    private void setupRowDeleting() {
        deleteBtn.setOnAction(event -> {
            HashMap<String,Object> selectedBook = tableView.getSelectionModel().getSelectedItem();
            if(selectedBook != null && bookDAO.deleteBook((String) selectedBook.get("BISBN"))) {
                tableView.getItems().remove(selectedBook);
            }
        });
    }

    public void setupCreating() {
        createBtn.setOnAction(event -> {
            String title = titleField.getText();
            String author = authorField.getText();
            int price = Integer.parseInt(priceField.getText());
            String isbn = isbnField.getText();

            boolean success = bookDAO.createBook(isbn, title, price, author);
            System.out.println(success);
            if(success) {

                HashMap<String,Object> book = new HashMap<>();
                book.put("BISBN", isbn);
                book.put("BTITLE", title);
                book.put("BAUTHOR", author);
                book.put("BPRICE", price);

                tableView.getItems().add(book);

                titleField.clear();
                authorField.clear();
                priceField.clear();
                isbnField.clear();
            }
        });
    }

    private void updateBook(HashMap<String, Object> book, String column, Object newValue) {
        String bisbn = (String) book.get("BISBN"); // ISBN을 기준으로 업데이트 수행
        System.out.println(bisbn);
        if (bookDAO.updateBook(bisbn, column, String.valueOf(newValue))) { // DB 업데이트 시도
            book.put(column, newValue); // HashMap 업데이트
        }
    }
}
```

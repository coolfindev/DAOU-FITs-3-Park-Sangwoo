package first;

import java.sql.*;
import java.util.Scanner;

public class Problem {
    public static void main(String[] args) {
        Connection conn = null;
        try {
            //오라클 드라이버 등록
            Class.forName("oracle.jdbc.driver.OracleDriver");

            //DBMS 연결
            String jdbcURL = "jdbc:oracle:thin:@localhost:1521:xe";
            String id = "C##DEV";
            String pw = "1423";
            conn = DriverManager.getConnection(jdbcURL, id, pw);

            //Statement 생성, 효율적인 확장을 위해 PreparedStatement로 작성
            String sql = "select bisbn, btitle, bauthor, bprice from book where btitle like ?";
            PreparedStatement ps = conn.prepareStatement(sql);

            //특정 키워드를 입력받기 위한 사용자 입력 받기
            Scanner sc = new Scanner(System.in);
            System.out.print("원하는 책 제목을 입력하세요 : ");
            ps.setString(1, "%"+ sc.nextLine() + "%");

            //Read를 위한 executeQuery()
            ResultSet rs = ps.executeQuery();
            while(rs.next()) {
                System.out.println(rs.getString("bisbn") + " " + rs.getString("btitle") + " " + rs.getString("bauthor") + " " + rs.getDouble("bprice"));
            }

            //연결 해제
            rs.close();
            ps.close();
            conn.close();
        } catch(ClassNotFoundException e) {
            System.out.println("클래스가 없습니다.");
        } catch (SQLException e) {
            System.out.println("DB와 연결에 실패했습니다.");
        }
    }
}

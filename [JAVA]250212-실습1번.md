package booksearchmvc.controller;

import booksearchmvc.service.BookSearchMVCService;
import booksearchmvc.service.BookSearchMVCServiceOracleImpl;
import booksearchmvc.vo.BookVO;
import javafx.collections.ObservableList;
import javafx.fxml.FXML;
import javafx.fxml.Initializable;
import javafx.scene.control.*;
import javafx.scene.control.cell.PropertyValueFactory;
import javafx.scene.control.cell.TextFieldTableCell;
import javafx.util.converter.IntegerStringConverter;

import java.net.URL;
import java.sql.Connection;
import java.sql.DriverManager;
import java.sql.PreparedStatement;
import java.sql.SQLException;
import java.util.ResourceBundle;

public class BookSearchMVCController implements Initializable {

    @FXML private TextField textField;
    @FXML private Button searchBtn;
    @FXML private Button deleteBtn;
    @FXML private TableView<BookVO> tableView;
    @FXML private TableColumn<BookVO, String> isbnCol;
    @FXML private TableColumn<BookVO, String> titleCol;
    @FXML private TableColumn<BookVO, Integer> priceCol;
    @FXML private TableColumn<BookVO, String> authorCol;

    String deleteIsbn;
    @Override
    public void initialize(URL url, ResourceBundle resourceBundle) {
        //View의 Controls같은 이벤트를 등록
        //refernece가 있어야

        isbnCol.setCellValueFactory(new PropertyValueFactory<>("bisbn"));
        titleCol.setCellValueFactory(new PropertyValueFactory<>("btitle"));
        priceCol.setCellValueFactory(new PropertyValueFactory<>("bprice"));
        authorCol.setCellValueFactory(new PropertyValueFactory<>("bauthor"));


        //수정 가능하게 만듦
        isbnCol.setCellFactory(TextFieldTableCell.forTableColumn());
        titleCol.setCellFactory(TextFieldTableCell.forTableColumn());
        authorCol.setCellFactory(TextFieldTableCell.forTableColumn());
        priceCol.setCellFactory(TextFieldTableCell.forTableColumn(new IntegerStringConverter()));
        tableView.setEditable(true);

        //수정 로직
        isbnCol.setOnEditCommit(event -> {
            BookVO book = event.getRowValue();
            String newIsbn = event.getNewValue();
            try {
                updateBookInDB(book.getBisbn(), "bisbn", newIsbn);
            } catch (Exception e) {
                throw new RuntimeException(e);
            }
            book.setBisbn(newIsbn);
        });

        titleCol.setOnEditCommit(event -> {
            BookVO book = event.getRowValue();
            String newTitle = event.getNewValue();
            try {
                updateBookInDB(book.getBisbn(), "btitle", newTitle);
            } catch (Exception e) {
                throw new RuntimeException(e);
            }
            book.setBtitle(newTitle);
        });

        authorCol.setOnEditCommit(event -> {
            BookVO book = event.getRowValue();
            String newAuthor = event.getNewValue();
            try {
                updateBookInDB(book.getBisbn(), "bauthor", newAuthor);
            } catch (Exception e) {
                throw new RuntimeException(e);
            }
            book.setBauthor(newAuthor);
        });

        priceCol.setOnEditCommit(event -> {
            BookVO book = event.getRowValue();
            Integer newPrice = event.getNewValue();
            try {
                updateBookInDB(book.getBisbn(), "bprice", String.valueOf(newPrice));
            } catch (Exception e) {
                throw new RuntimeException(e);
            }
            book.setBprice(newPrice);
        });

        tableView.setRowFactory(e -> {
            TableRow<BookVO>  row = new TableRow<>();
            row.setOnMouseClicked(event -> {
                if (event.getClickCount() == 1) {
                    BookVO book = row.getItem();
                    deleteIsbn = book.getBisbn();
                }
            });
            return row;
        });

        searchBtn.setOnAction(event -> {
            BookSearchMVCService service = new BookSearchMVCServiceOracleImpl();
            ObservableList<BookVO> list = null;
            try {
                list = service.searchBookByKeyword(textField.getText());
            } catch (Exception e) {
                throw new RuntimeException(e);
            }
            tableView.setItems(list);
        });

        deleteBtn.setOnAction(event -> {
            BookSearchMVCService service = new BookSearchMVCServiceOracleImpl();
            ObservableList<BookVO> list = null;
            try {
                list = service.deleteSelectedBook(deleteIsbn, textField.getText());
            } catch (Exception e) {
                throw new RuntimeException(e);
            }
            tableView.setItems(list);
        });
    }
    private void updateBookInDB(String bisbn, String column, String newValue) throws Exception{
        String sql = "UPDATE book SET " + column + " = ? WHERE bisbn = ?";
        Connection con = DriverManager.getConnection("jdbc:oracle:thin:@localhost:1521:xe", "C##DEV", "1423");
        PreparedStatement ps = con.prepareStatement(sql);

        if (column.equals("bprice")) {
            ps.setInt(1, Integer.parseInt(newValue)); // 가격은 숫자로 변환
        } else {
            ps.setString(1, newValue);
        }
        ps.setString(2, bisbn); // WHERE 조건에 기존 ISBN 사용

        int affectedRows = ps.executeUpdate();
        if (affectedRows > 0) {
            System.out.println(column + " 업데이트 성공: " + newValue);
        } else {
            System.out.println(column + " 업데이트 실패");
        }


    }
}

package booksearchmvc.dao;

import booksearchmvc.vo.BookVO;
import javafx.collections.FXCollections;
import javafx.collections.ObservableList;

import java.sql.*;

public class BookDAO {

    Connection conn = null;
    public BookDAO(Connection conn) {
        this.conn = conn;
    }

    public ObservableList<BookVO> select(String text) {
        ObservableList<BookVO> books = FXCollections.observableArrayList();
        try {
            String sql = "select bisbn, btitle, bprice, bauthor ";
            sql += "from book where btitle like ?";

            PreparedStatement ps = conn.prepareStatement(sql);
            ps.setString(1, "%" + text + "%");

            ResultSet rs = ps.executeQuery();
            while (rs.next()) {
                BookVO book = new BookVO(rs.getString("bisbn"), rs.getString("btitle"), rs.getInt("bprice"), rs.getString("bauthor"));
                books.add(book);
            }
            rs.close();
            ps.close();
        } catch (SQLException e) {
            e.printStackTrace();
        }
        return books;
    }

    public boolean delete(Connection conn, String bisbn) throws SQLException {
        String sql = "DELETE FROM book WHERE bisbn = ?";
        try (PreparedStatement ps = conn.prepareStatement(sql)) {
            ps.setString(1, bisbn);
            return ps.executeUpdate() > 0;
        }
    }
    public boolean update(Connection conn, String bisbn, String column, String newValue) throws SQLException {
        String sql = "UPDATE book SET " + column + " = ? WHERE bisbn = ?";
        try (PreparedStatement ps = conn.prepareStatement(sql)) {
            if (column.equals("bprice")) {
                ps.setInt(1, Integer.parseInt(newValue));
            } else {
                ps.setString(1, newValue);
            }
            ps.setString(2, bisbn);
            return ps.executeUpdate() > 0;
        }
    }

}

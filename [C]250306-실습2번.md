### 2p 문제

```c
#define _CRT_SECURE_NO_WARNINGS
#include <stdio.h>

int main() {
    
    for (int i = 4; i > 0; i--) {
        for (int j = 0; j < i; j++) {
            printf("*");
        }
        printf("\n");
    }
    return 0;
}
```

### 3p 문제

```c
#define _CRT_SECURE_NO_WARNINGS
#include <stdio.h>

#define _USE_MATH_DEFINES
#include <math.h>
int main() {
    int n, r;
    do {
        printf("========================\n1. 원의 둘레 구하기\n2. 원의 넓이 구하기\n3. 구의 부피 구하기\n4. 그만두기\n========================\n");
        scanf_s("%d", &n);

        switch (n) {
            case 1:
                printf(">> 반지름은? ");
                scanf_s("%d", &r);

                printf(">> 반지름이 %d인 원의 둘레는 %.2lf\n\n", r, 2 * r * M_PI);
                break;
            case 2:
                printf(">> 반지름은? ");
                scanf_s("%d", &r);

                printf(">> 반지름이 %d인 원의 넓이는 %.2lf\n\n", r, r * r * M_PI);
                break;
            case 3:
                printf(">> 반지름은? ");
                scanf_s("%d", &r);

                printf(">> 반지름이 %d인 구의 부피는 %.2lf\n\n", r, 3/4 * r * r * r * M_PI);
                break;
        }
    } while (n != 4);
}

```

### 프로젝트 주제

**증권 주식 계좌 및 자산관리 시스템**

### 프로젝트 개요

- 주제 선정 배경(최선집행기준)
    - 증권사는 최선집행의무가 있으며 고객들은 해당 기준에 따라 거래가 이루어 지는지 확인할 권리가 있음
    - 기존 SOR 알고리즘 구현을 통해 다중 거래소 거래 환경을 구현을 진행했지만 위의 내용은 고려하지 않았었음
    - 이번 프로젝트를 통해 최선집행의무에 따른 거래소 산정 근거를 보고서 형태로 저장/관리함으로써 현업에 도움이 될 것
- 주요 기능
    - ATS 거래( sor 시스템, 알고리즘)
        - 대체거래소 가격을 랜덤으로 할지, api 사용이 가능한지
        - 매수/매도
        - 구매 내역(포트폴리오)
        - 최종 보고서(거래소 선택 이유)⇒ 실제로 거래 발생시 선택의 이유를 제출 해야함
    - 개인
        - 매수 매도
        - 종목 상세
        - 개인 보고서
    - 관리자
        - 전체 거래 내역
        - 개인별 내역
        - (휴먼계, 신고)

- 개발

---

- 매수/매도 ← 대균
- SOR 알고리즘 ← 강온
- 보고서 (JSON)  ← 상우
- 체결 미체결 ← 대균
- 잔고 X

- 문서

---

- 사용자 지침서 ← 다같이
- 요구사항 정의서 ←  강온
- PPT ← 상우

void exportJson(int n) {
    //0 : 일반사용자
    //1 : 관리자
    set_env();

    char* select_sql = "SELECT CHARGE, OFFERNUMBER, EXCHANGEACTUAL, TRADENUMBER, QUANTITY, PRICE, CREATEDAT, SELLERACCOUNTNUMBER, BUYERACCOUNTNUMBER FROM trades";

    OCIHandleAlloc(envhp, (void**)&stmthp, OCI_HTYPE_STMT, 0, NULL);
    OCIStmtPrepare(stmthp, errhp, (text*)select_sql, strlen(select_sql), OCI_NTV_SYNTAX, OCI_DEFAULT);

    if (OCIStmtExecute(svchp, stmthp, errhp, 0, 0, NULL, NULL, OCI_DEFAULT) != OCI_SUCCESS) {
        quit_env();
        cls;
        printf("데이터 불러오기 실패!\n");
    }
    else {
        printf("데이터 저장 중...\n");
        Sleep(1000);
        
        double charge;
        long  offerNumber, exchangeActual, tradeNumber, quantity, price;
        char createdAt[30], sellerAccount[64], buyerAccount[64];

        OCIDefine* def1 = NULL, * def2 = NULL, * def3 = NULL, * def4 = NULL, * def5 = NULL, * def6 = NULL, * def7 = NULL, * def8 = NULL, * def9 = NULL;

        OCIDefineByPos(stmthp, &def1, errhp, 1, &charge, sizeof(charge), SQLT_FLT, NULL, NULL, NULL, OCI_DEFAULT);
        OCIDefineByPos(stmthp, &def2, errhp, 2, &offerNumber, sizeof(offerNumber), SQLT_INT, NULL, NULL, NULL, OCI_DEFAULT);
        OCIDefineByPos(stmthp, &def3, errhp, 3, &exchangeActual, sizeof(exchangeActual), SQLT_INT, NULL, NULL, NULL, OCI_DEFAULT);
        OCIDefineByPos(stmthp, &def4, errhp, 4, &tradeNumber, sizeof(tradeNumber), SQLT_INT, NULL, NULL, NULL, OCI_DEFAULT);
        OCIDefineByPos(stmthp, &def5, errhp, 5, &quantity, sizeof(quantity), SQLT_INT, NULL, NULL, NULL, OCI_DEFAULT);
        OCIDefineByPos(stmthp, &def6, errhp, 6, &price, sizeof(price), SQLT_INT, NULL, NULL, NULL, OCI_DEFAULT);
        OCIDefineByPos(stmthp, &def7, errhp, 7, createdAt, sizeof(createdAt), SQLT_STR, NULL, NULL, NULL, OCI_DEFAULT);
        OCIDefineByPos(stmthp, &def8, errhp, 8, sellerAccount, sizeof(sellerAccount), SQLT_STR, NULL, NULL, NULL, OCI_DEFAULT);
        OCIDefineByPos(stmthp, &def9, errhp, 9, buyerAccount, sizeof(buyerAccount), SQLT_STR, NULL, NULL, NULL, OCI_DEFAULT);

        FILE* file = fopen("C:/coolfin/projc/public/user_data.json", "w");
        if (file == NULL) {
            printf("파일을 열 수 없습니다.\n");
            quit_env();
            return;
        }

        fprintf(file, "{\n    \"trades\": [\n");

        int first = 1; // 첫 번째 데이터인지 확인하는 플래그

        while ((status = OCIStmtFetch2(stmthp, errhp, 1, OCI_DEFAULT, 0, OCI_DEFAULT)) == OCI_SUCCESS || status == OCI_SUCCESS_WITH_INFO) {
            if (!first) {
                fprintf(file, ",\n");
            }
            first = 0;

            fprintf(file, "        {\n");
            fprintf(file, "            \"tradeNumber\": %ld,\n", tradeNumber);
            fprintf(file, "            \"charge\": %.6f,\n", charge);
            fprintf(file, "            \"offerNumber\": %ld,\n", offerNumber);
            fprintf(file, "            \"exchangeActual\": %ld,\n", exchangeActual);
            fprintf(file, "            \"quantity\": %ld,\n", quantity);
            fprintf(file, "            \"price\": %ld,\n", price);
            fprintf(file, "            \"createdAt\": \"%s\",\n", createdAt);
            fprintf(file, "            \"sellerAccountNumber\": \"%s\",\n", sellerAccount);
            fprintf(file, "            \"buyerAccountNumber\": \"%s\"\n", buyerAccount);
            fprintf(file, "        }");
        }

        // JSON 닫기
        fprintf(file, "\n    ]\n}\n");
        fclose(file);
        quit_env();

        printf("데이터가 JSON 파일로 저장되었습니다.\n");

        printf("계속하시려면 아무 키나 입력해 주세요...");
        pause;
        cls;
    }

void login() {
    set_env();
	printLogo();
	displayWelcomeMessage();
    char username[50];
    char password[50];

    // 로그인 입력 받기
    printf("아이디를 입력하세요: ");
    scanf("%s", username);

    printf("비밀번호를 입력하세요: ");
    int i = 0;
    while (1) {
        char ch = getch();
        if (ch == 13) {      // Enter 키를 누르면 입력 종료
            break;
        }
        if (ch == 8 && i > 0) {
            i--;
            printf("\b \b");
        }
        else {
            password[i++] = ch;
            printf("*");  // 입력된 문자는 화면에 '*'로 표시
        }
    }

    password[i] = '\0';  // 문자열 종료

    char* select_sql = "SELECT userid, name, type FROM USERS WHERE userid = :1 AND password = :2";
    OCIHandleAlloc(envhp, (void**)&stmthp, OCI_HTYPE_STMT, 0, NULL);
    OCIStmtPrepare(stmthp, errhp, (text*)select_sql, strlen(select_sql), OCI_NTV_SYNTAX, OCI_DEFAULT);
    //OCIStmtExecute(svchp, stmthp, errhp, 0, 0, NULL, NULL, OCI_DEFAULT);

    //바인딩
    OCIBind* bind1 = NULL, * bind2 = NULL;
    OCIBindByPos(stmthp, &bind1, errhp, 1, username, sizeof(username), SQLT_STR, NULL, NULL, NULL, 0, NULL, OCI_DEFAULT);
    OCIBindByPos(stmthp, &bind2, errhp, 2, password, sizeof(password), SQLT_STR, NULL, NULL, NULL, 0, NULL, OCI_DEFAULT);

	if (OCIStmtExecute(svchp, stmthp, errhp, 0, 0, NULL, NULL, OCI_DEFAULT) != OCI_SUCCESS) {
        quit_env();
        cls;
        printf("로그인 실패! 아이디 또는 비밀번호를 확인해 주세요.\n");
        //check_error(errhp);
        login(stmthp, errhp, svchp);
    }

    else {
        char name[50], userId[50], type[20];

        OCIDefineByPos(stmthp, &def1, errhp, 1, userId, sizeof(userId), SQLT_STR, NULL, NULL, NULL, OCI_DEFAULT);
        OCIDefineByPos(stmthp, &def1, errhp, 2, name, sizeof(name), SQLT_STR, NULL, NULL, NULL, OCI_DEFAULT);
        OCIDefineByPos(stmthp, &def1, errhp, 3, type, sizeof(type), SQLT_STR, NULL, NULL, NULL, OCI_DEFAULT);
